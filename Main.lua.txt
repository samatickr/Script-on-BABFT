local function makeToggle(parent, initialState, callback)
    local Btn = Instance.new("TextButton")
    Btn.Size = UDim2.new(0, 80, 0, 26)
    Btn.Position = UDim2.new(1, -88, 0.5, -13)
    Btn.BorderSizePixel = 0
    Btn.Text = initialState and "Enable" or "Disable"
    Btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    Btn.Font = Enum.Font.GothamBold
    Btn.TextSize = 12
    Btn.BackgroundColor3 = initialState and Color3.fromRGB(60, 180, 60) or Color3.fromRGB(180, 50, 50)
    Btn.Parent = parent
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, 5)
    c.Parent = Btn
    local state = initialState
    Btn.MouseButton1Click:Connect(function()
        state = not state
        Btn.Text = state and "Enable" or "Disable"
        Btn.BackgroundColor3 = state and Color3.fromRGB(60, 180, 60) or Color3.fromRGB(180, 50, 50)
        if callback then callback(state) end
    end)
    return Btn
end

        local num = tonumber(Box.Text)

            Box.TextColor3 = Color3.fromRGB(255, 80, 80)

    b.BackgroundTransparency = 0.15

    b.Font = Enum.Font.GothamBold
    b.TextSize = 24
    b.AutoButtonColor = false
    b.Parent = FlyControls
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, 8)
    c.Parent = b
    local s = Instance.new("UIStroke")
    s.Color = Color3.fromRGB(0, 0, 0)
    s.Thickness = 2
    s.Parent = b
    local function press()
        flyKeys[key] = true
        b.BackgroundColor3 = Color3.fromRGB(200, 170, 0)
    end
    local function release()
        flyKeys[key] = false
        b.BackgroundColor3 = Color3.fromRGB(255, 220, 0)
    end

    b.MouseButton1Up:Connect(release)
    b.MouseLeave:Connect(release)
    b.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then press() end
    end)
    b.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then release() end
    end)
    return b

task.spawn(function()
    while true do
        task.wait()
        if autofarm then
            if not HRP or not HRP.Parent then 
                character = LP.Character
                if character then HRP = character:WaitForChild("HumanoidRootPart", 5) end
            end
            
            if HRP and HRP.Parent then
                if farmIndex == 11 then
                    local Stages = Workspace:FindFirstChild("BoatStages")
                    if Stages then
                        local normalStages = Stages:FindFirstChild("NormalStages")
                        if normalStages then
                            local endpoint = normalStages:FindFirstChild("TheEnd")
                            if endpoint then
                                local chest = endpoint:FindFirstChild("GoldenChest")
                                if chest then
                                    HRP:PivotTo(chest:GetPivot() + Vector3.new(0, 0, -10))
                                    local ii = 0
                                    repeat
                                        task.wait(1)
                                        ii += 1
                                        if ii % 20 == 0 and HRP then
                                            HRP:PivotTo(chest:GetPivot() + Vector3.new(0, 0, -10))
                                        end
                                        if not HRP or not HRP.Parent then break end
                                    until (HRP.Position - chest:GetPivot().Position).Magnitude > 500 or not autofarm
                                    farmIndex = 1
                                end
                            end
                        end
                    end
                else
                    local stages = Workspace:FindFirstChild("BoatStages")
                    if stages then
                        local normalStages = stages:FindFirstChild("NormalStages")
                        if normalStages then
                            local roomName = "CaveStage" .. farmIndex
                            local stage = normalStages:FindFirstChild(roomName)
                            if stage then
                                local darkPart = stage:FindFirstChild("DarknessPart")
                                if darkPart then
                                    if character then character:PivotTo(darkPart.CFrame - Vector3.new(0, 0, 15)) end
                                    local tween2 = TweenService:Create(HRP, TweenInfo.new(2, Enum.EasingStyle.Linear), {CFrame = darkPart.CFrame + Vector3.new(0, 0, 20)})
                                    tweening = true
                                    tween2:Play()
                                    tween2.Completed:Wait()
                                    tweening = false
                                    farmIndex += 1
                                end
                            end
                        end
                    end
                end
            end
        end
    end
end)

        HRP.Velocity = Vector3.zero
    end
end)

local function shakeMenu()
    local originalPos = MainFrame.Position
    local shakeOffsets = {
        UDim2.new(0, 8, 0, 0), UDim2.new(0, -8, 0, 0), UDim2.new(0, 0, 0, 8), UDim2.new(0, 0, 0, -8),
        UDim2.new(0, 6, 0, 6), UDim2.new(0, -6, 0, -6), UDim2.new(0, 6, 0, -6), UDim2.new(0, -6, 0, 6),
    }
    for i = 1, 12 do
        local offset = shakeOffsets[(i - 1) % #shakeOffsets + 1]
        MainFrame.Position = UDim2.new(
            originalPos.X.Scale, originalPos.X.Offset + offset.X.Offset,
            originalPos.Y.Scale, originalPos.Y.Offset + offset.Y.Offset
        )
        task.wait(0.04)
    end
    MainFrame.Position = originalPos
end

    local char = LP.Character

    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    hum.WalkSpeed = SPEED_VALUE
    hum.UseJumpPower = true
    hum.JumpPower = JUMP_VALUE
end

    task.wait(1)
    applyAll()
end)

RunService.Stepped:Connect(function()
    if not noclipEnabled then return end
    local char = LP.Character
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.CanCollide then
            part.CanCollide = false
        end
    end
end)

RunService.RenderStepped:Connect(function(dt)
    if not flyEnabled then return end
    local char = LP.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    hum.PlatformStand = true
    hrp.Velocity = Vector3.zero
    hrp.RotVelocity = Vector3.zero

    if not lastPos then lastPos = hrp.Position end

    local camCF = camera.CFrame
    local moveDir = Vector3.zero

    if flyKeys.W then moveDir = moveDir + camCF.LookVector end
    if flyKeys.S then moveDir = moveDir - camCF.LookVector end
    if flyKeys.A then moveDir = moveDir - camCF.RightVector end
    if flyKeys.D then moveDir = moveDir + camCF.RightVector end
    if flyKeys.Space then moveDir = moveDir + Vector3.new(0, 1, 0) end
    if flyKeys.Ctrl then moveDir = moveDir - Vector3.new(0, 1, 0) end

    local newPos
    if moveDir.Magnitude > 0 then
        newPos = lastPos + moveDir.Unit * FLY_SPEED * dt * 3
    else
        newPos = lastPos
    end

    hrp.CFrame = CFrame.new(newPos, newPos + camCF.LookVector)
    lastPos = newPos
end)
