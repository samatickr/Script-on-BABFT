-- ========================================================
-- 1. ПОЛУЧАЕМ МЕТАТАБЛИЦУ (для перехвата)
-- ========================================================
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
local oldIndex = mt.__index
setreadonly(mt, false)

-- ========================================================
-- 2. ПЕРЕХВАТ ВЫЗОВОВ (Kick, HttpGet, Request)
-- ========================================================
mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    -- АНТИ-КИК
    if method == "Kick" and self == LocalPlayer then
        warn("[ANTI-KICK] Кик заблокирован!")
        return nil
    end

    -- ПОДМЕНА HTTP ЗАПРОСОВ (HttpGet / HttpGetAsync)
    if (method == "HttpGet" or method == "HttpGetAsync") and self == game then
        local url = args[1]
        if type(url) == "string" then
            local lowerUrl = string.lower(url)
            if string.find(lowerUrl, "key") or string.find(lowerUrl, "validate") or string.find(lowerUrl, "butter") or string.find(lowerUrl, "auth") then
                warn("[BYPASS] Перехвачен HttpGet: " .. url)
                return '"true"' -- Подсовываем успешный ответ
            end
        end
    end

    return oldNamecall(self, ...)
end)

-- Защита от чтения свойства Kick как функции
mt.__index = newcclosure(function(self, key)
    if self == LocalPlayer and key == "Kick" then
        return function() end 
    end
    return oldIndex(self, key)
end)

setreadonly(mt, true)

-- ========================================================
-- 3. ПЕРЕХВАТ REQUEST (Если скрипт использует syn.request)
-- ========================================================
local oldRequest = request or http_request or (syn and syn.request)
if oldRequest then
    local function newRequest(options)
        if options and options.Url and type(options.Url) == "string" then
            local lowerUrl = string.lower(options.Url)
            if string.find(lowerUrl, "key") or string.find(lowerUrl, "validate") or string.find(lowerUrl, "butter") or string.find(lowerUrl, "auth") then
                warn("[BYPASS] Перехвачен request: " .. options.Url)
                return {StatusCode = 200, Body = '"true"'}
            end
        end
        return oldRequest(options)
    end
    
    -- Безопасная подмена
    pcall(function() request = newRequest end)
    pcall(function() http_request = newRequest end)
    pcall(function() if syn then syn.request = newRequest end end)
end
