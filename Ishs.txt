-- 2. АНТИ-КИК ХУК (Блокирует вызов Player:Kick())
-- ========================================================
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

local oldKick = LocalPlayer.Kick
LocalPlayer.Kick = newcclosure(function(self, reason)
    if self == LocalPlayer then
        warn("[ANTI-KICK] Попытка кика заблокирована! Причина: " .. tostring(reason))
        return nil -- Игнорируем кик
    end
    return oldKick(self, reason)
end)

-- Дополнительный хук через метаметод (на случай смены метода)
local mt = getrawmetatable(game)
setreadonly(mt, false)
local oldNamecall = mt.__namecall
mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    if method == "Kick" and self == LocalPlayer then
        warn("[ANTI-KICK] Перехвачен Kick через __namecall!")
        return nil
    end
    return oldNamecall(self, ...)
end)
