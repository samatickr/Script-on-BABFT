-- 1. ПОДМЕНА ОТВЕТА СЕРВЕРА ПРОВЕРКИ КЛЮЧА
-- ========================================================
local oldHttpGet = game.HttpGet
game.HttpGet = function(self, url, ...)
    -- Если скрипт пытается проверить ключ (ищем по ссылке)
    if string.find(string.lower(url), "key") or string.find(string.lower(url), "validate") or string.find(string.lower(url), "butter") then
        warn("[BYPASS] Перехвачен запрос ключа: " .. url)
        -- Возвращаем фейковый успешный ответ
        return '"true"' 
    end
    return oldHttpGet(self, url, ...)
end

-- Перехват request (на случай если используется Synapse/Http)
local oldRequest = request or http_request or (syn and syn.request)
if oldRequest then
    local function newRequest(options)
        if options and options.Url and (string.find(string.lower(options.Url), "key") or string.find(string.lower(options.Url), "validate") or string.find(string.lower(options.Url), "butter")) then
            warn("[BYPASS] Перехвачен request ключа: " .. options.Url)
            return {StatusCode = 200, Body = '"true"'}
        end
        return oldRequest(options)
    end
    if syn and syn.request then syn.request = newRequest end
    if http and http.request then http.request = newRequest end
    if request then request = newRequest end
end


-- ========================================================
-- 2. КОМПЛЕКСНАЯ БЛОКИРОВКА КИКА (оставляем)
-- ========================================================
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
local oldIndex = mt.__index
setreadonly(mt, false)

mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    if method == "Kick" and self == LocalPlayer then
        warn("[ANTI-KICK] Вызов Kick через : заблокирован!")
        return nil
    end
    return oldNamecall(self, ...)
end)

mt.__index = newcclosure(function(self, key)
    if self == LocalPlayer and key == "Kick" then
        return function() end 
    end
    return oldIndex(self, key)
end)
setreadonly(mt, true)
